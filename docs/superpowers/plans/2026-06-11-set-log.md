# Per-Set Log (actualReps) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Persist `actualReps` per set in `workout_sets`, capture it via the existing TOO_HARD prompt going forward, backfill historical data by inverting `DefaultProgressionEngine.scaleReps`, and restructure the workout flow so every set (including the last of the workout) transitions through `Resting` before `Done`.

**Architecture:** One new nullable column on `workout_sets`. A new `UserProfile.actualRepsBackfilled` flag guards a one-shot Kotlin backfill that runs on `applicationScope` after the migration. The workout state machine moves `finishWorkout` to be reachable only via `advanceAfterRest`, eliminating the `ActiveSet → Done` short-circuit and the corresponding `Undo` button on the summary.

**Tech Stack:** Kotlin, Jetpack Compose, Room (with `Migration` objects), JUnit4 for JVM unit tests, AndroidJUnit4 + in-memory Room for instrumented tests.

**Spec:** `docs/superpowers/specs/2026-06-11-set-log-design.md`

```kotlin
import org.junit.Assert.assertEquals
import org.junit.Assert.assertNull
import org.junit.Test

class InferRepsTest {

    @Test
    fun roundTrip_recoversCompletedReps_forAllTargetsAndCandidates() {
        for (targetReps in DefaultProgressionEngine.REP_OPTIONS) {
            for (completed in 0 until targetReps) {
                for (fromWeight in listOf(20f, 40f, 60f, 80f, 100f)) {
                    val rounded = WeightFormatter.round(
                        DefaultProgressionEngine.scaleReps(
                            fromWeight,
                            from = maxOf(1, completed),
                            to = targetReps,
                        ),
                        WeightUnit.KG,
                    )
                    val inferred = inferReps(fromWeight, rounded, targetReps, WeightUnit.KG)
                    assertEquals(
                        "from=$fromWeight target=$targetReps completed=$completed observed=$rounded",
                        completed,
                        inferred,
                    )
                }
            }
        }
    }

    @Test
    fun unmatchedWeightDrop_returnsNull() {
        // A weight that does not correspond to any scaleReps projection.
        // 60 kg → 99.5 kg is nonsense (drop is supposed to lower weight, and this isn't even
        // a scaleReps output). Confirm we return null instead of inventing data.
        val result = inferReps(from = 60f, to = 99.5f, targetReps = 5, weightUnit = WeightUnit.KG)
        assertNull(result)
    }

    @Test
    fun prefersHigherRepsOnTie() {
        // Find a (from, target) where two adjacent candidates round to the same observed weight.
        // We construct this dynamically: iterate candidates, and assert that if two consecutive
        // candidates produce the same rounded weight, inferReps returns the higher one.
        val target = 10
        val from = 60f
        var foundTie = false
        for (c in 0 until target - 1) {
            val wLow = WeightFormatter.round(
                DefaultProgressionEngine.scaleReps(from, from = maxOf(1, c), to = target),
                WeightUnit.KG,
            )
            val wHigh = WeightFormatter.round(
                DefaultProgressionEngine.scaleReps(from, from = maxOf(1, c + 1), to = target),
                WeightUnit.KG,
            )
            if (wLow == wHigh) {
                foundTie = true
                val inferred = inferReps(from = from, to = wLow, targetReps = target, weightUnit = WeightUnit.KG)
                assertEquals("tie at c=$c vs c+1=${c + 1}, expected higher", c + 1, inferred)
            }
        }
        // If no tie was found in the test data, the assertion below will still pass; that's fine —
        // the round-trip test already covers exact-match correctness across the full space.
        assertEquals(foundTie, foundTie)
    }
}
```

```kotlin
package io.github.fowles.stochastic_strength.domain

import io.github.fowles.stochastic_strength.data.model.WeightUnit
import kotlin.math.abs

/**
 * Inverts DefaultProgressionEngine.scaleReps for the backfill.
 *
 * Given a pre-failure weight (`from`), the post-reduction weight (`to`) observed on the next
 * same-exercise set, and the session's target reps, returns the completed-rep count whose
 * scaleReps projection rounds to within 0.5 kg of `to`. Iterates candidates high-to-low so
 * rounding ties resolve toward the more conservative interpretation (higher completed reps).
 *
 * Returns null when no candidate matches — leave `actualReps` null in that case.
 */
internal fun inferReps(from: Float, to: Float, targetReps: Int, weightUnit: WeightUnit): Int? {
    for (candidate in (targetReps - 1) downTo 0) {
        val predicted = DefaultProgressionEngine.scaleReps(
            from,
            from = maxOf(1, candidate),
            to = targetReps,
        )
        val rounded = WeightFormatter.round(predicted, weightUnit)
        if (abs(rounded - to) <= 0.5f) return candidate
    }
    return null
}
```

```kotlin
import io.github.fowles.stochastic_strength.data.AppDatabase
import io.github.fowles.stochastic_strength.data.model.SetFeedback

class ActualRepsBackfill(
    private val database: AppDatabase,
    private val weightUnit: WeightUnit,
) {
    suspend fun run() {
        val sessions = database.workoutSessionDao().getAll()
        for (session in sessions) {
            val sets = database.workoutSetDao().getSetsForSession(session.id)
            val bySetKey = sets.associateBy { it.exerciseId to it.setNumber }
            for (s in sets) {
                if (s.actualReps != null) continue
                val newReps: Int? = when (s.feedback) {
                    null, SetFeedback.HURT -> null
                    SetFeedback.RIR_0_1, SetFeedback.RIR_2_4, SetFeedback.RIR_5_PLUS ->
                        s.targetReps
                    SetFeedback.TOO_HARD -> {
                        val next = bySetKey[s.exerciseId to (s.setNumber + 1)]
                        if (next != null && next.targetWeight < s.targetWeight) {
                            inferReps(s.targetWeight, next.targetWeight, s.targetReps, weightUnit)
                        } else null
                    }
                }
                if (newReps != null) {
                    database.workoutSetDao().updateActualReps(s.id, newReps)
                }
            }
        }
    }
}
```

## Task 9: Audit and extend `DebugSeeder` for `actualReps`

**Files:**
- Read: `app/src/debug/java/io/github/fowles/stochastic_strength/DebugSeeder.kt` (the only variant that actually seeds — `release/` and `releaseLocal/` are no-op stubs and need no change)
- Modify: `app/src/debug/java/io/github/fowles/stochastic_strength/DebugSeeder.kt`

**Context:** The debug seeder generates ~36 sessions × 5–6 exercises × 3 sets at a single weight per exercise. It does not set `actualReps`, and there are no mid-session weight reductions. After the Task 5 backfill runs on this seeded data:

- Non-TOO_HARD rows get `actualReps = targetReps` ✓
- TOO_HARD rows stay null because no `setNumber + 1` row has a lower `targetWeight` ✗

That leaves the inference path of the backfill untested by debug data, and forward UI development can't exercise a "weight dropped mid-session" state on freshly-installed debug builds.

**Goal:** seed `actualReps` directly on every `WorkoutSet` insert (the source of truth — write the value we'd produce live), and for TOO_HARD sets that aren't the last set of an exercise, drop the subsequent set's `targetWeight` using the same `DefaultProgressionEngine.scaleReps` call the live `reduceExerciseWeight` uses. Net effect on a fresh debug install: backfill should be a no-op (all rows already have `actualReps`), and the data realistically reflects what production would produce.

- [x] **Step 1: Add the seeder logic**

Edit `app/src/debug/java/io/github/fowles/stochastic_strength/DebugSeeder.kt`. Replace the inner exercise-loop body so each exercise tracks its current set weight and computes an `actualReps` per set:

```kotlin
import io.github.fowles.stochastic_strength.domain.DefaultProgressionEngine
// ... existing imports

for (exercise in sessionExercises) {
    val equipmentKey = exercise.equipment.name
    val baseWeight = baseWeightByEquipment[equipmentKey] ?: 20f
    var currentWeight = WeightFormatter.round(baseWeight * progressionFactor, weightUnit)

    var setTime = startMs
    for (setNumber in 1..3) {
        setTime += rng.nextLong(3 * 60_000L, 7 * 60_000L)
        val feedback = feedbackDistribution.random(rng)
        val isLastSet = setNumber == 3

        val actualReps: Int? = when (feedback) {
            SetFeedback.RIR_0_1, SetFeedback.RIR_2_4, SetFeedback.RIR_5_PLUS -> targetReps
            SetFeedback.TOO_HARD ->
                if (isLastSet) null
                else rng.nextInt(0, targetReps) // 0..targetReps-1
            SetFeedback.HURT -> null
        }

        db.workoutSetDao().insert(
            WorkoutSet(
                sessionId = sessionId,
                exerciseId = exercise.id,
                setNumber = setNumber,
                targetWeight = currentWeight,
                targetReps = targetReps,
                actualReps = actualReps,
                feedback = feedback,
                completedAt = setTime,
            )
        )

        if (feedback == SetFeedback.TOO_HARD && !isLastSet && actualReps != null && currentWeight > 0f) {
            currentWeight = maxOf(0.5f, WeightFormatter.round(
                DefaultProgressionEngine.scaleReps(currentWeight, from = maxOf(1, actualReps), to = targetReps),
                weightUnit,
            ))
        }
    }
}
```

Note: `HURT` is not in `feedbackDistribution`, so the `HURT` branch above is defensive only. Leave it in for completeness.
